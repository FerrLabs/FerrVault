     name: prod

     url: https://ferrvault.example.com
     organization: your-org-slug
     tokenSecretRef:
       name: ferrvault-api-token
       key: token

   apiVersion: ferrvault.com/v1alpha1
   kind: FerrVaultSecret
   metadata:
     name: web-env
     namespace: default
   spec:
     connectionRef: { name: prod }
     project: web
     vault: production
     selector:
       names: [DATABASE_URL, STRIPE_KEY]

   kubectl get ferrvaultsecrets,secrets -n default
   kubectl describe ferrvaultsecret web-env -n default
